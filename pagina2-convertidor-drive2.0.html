var FOLDER_ID = "1aRJRe0623pBilxj7AqVJYsmDgi-LqwzY"; var SHEET_ID = "1NV3p9HJLZFZJLOkj8R2fXKnphIXM8KRP1c8ctZUUV3A"; // ID de tu Google Sheet // ---------------------------------------------------- // GET: búsqueda de archivos por folio o validación en Sheet // ---------------------------------------------------- function doGet(e) { try { var folio = e.parameter.folio; var action = e.parameter.action; if (!folio) { return jsonOutput({ found: false, message: "Folio no proporcionado" }); } // NUEVA FUNCIÓN: Verificar si el folio existe en Google Sheets para el HTML Front if (action === 'checkFolio') { var sheet = SpreadsheetApp.openById(SHEET_ID).getActiveSheet(); var data = sheet.getRange("A:A").getValues(); var existe = false; for (var i = 0; i < data.length; i++) { if (data[i][0].toString() === folio) { existe = true; break; // Detiene la búsqueda en el primer match } } return jsonOutput({ exists: existe }); } // ORIGINAL: Búsqueda de archivos en Drive var folder = DriveApp.getFolderById(FOLDER_ID); var result = { found: false, message: "No se encontraron archivos para el folio " + folio }; var folioSafe = folio.replace(/'/g, "\\'"); // Buscar INE var ineQuery = "'" + FOLDER_ID + "' in parents and title contains '" + folioSafe + "-ine' and trashed = false"; Logger.log("Query INE: " + ineQuery); var ineIt = folder.searchFiles(ineQuery); if (ineIt.hasNext()) { var ineFile = ineIt.next(); result.ineBase64 = Utilities.base64Encode(ineFile.getBlob().getBytes()); result.found = true; } // Buscar Selfie var selfieQuery = "'" + FOLDER_ID + "' in parents and title contains '" + folioSafe + "-foto' and trashed = false"; Logger.log("Query Selfie: " + selfieQuery); var selfieIt = folder.searchFiles(selfieQuery); if (selfieIt.hasNext()) { var selfieFile = selfieIt.next(); result.selfieBase64 = Utilities.base64Encode(selfieFile.getBlob().getBytes()); result.found = true; } if (result.found) { result.message = "Archivos encontrados para el folio " + folio; } return jsonOutput(result); } catch (error) { return jsonOutput({ found: false, message: "Error: " + error.toString() }); } } // ---------------------------------------------------- // POST: subida de imágenes y actualización en Sheets // ---------------------------------------------------- function doPost(e) { try { var data = JSON.parse(e.postData.contents); // NUEVA FUNCIÓN: Actualizar Google Sheets cuando el Front mande el aviso if (data.action === "updateSheet") { var folio = data.folio; var sheet = SpreadsheetApp.openById(SHEET_ID).getActiveSheet(); var sheetData = sheet.getDataRange().getValues(); for (var i = 1; i < sheetData.length; i++) { // i=1 omite la fila 1 (encabezados) if (sheetData[i][0].toString() === folio) { var rowIndex = i + 1; // Fila real en la hoja (Base 1) // Columna B (Índice 1): folio-ine.jpg if (sheetData[i][1] === "No" || sheetData[i][1] === "") { sheet.getRange(rowIndex, 2).setValue("Si"); } // Columna C (Índice 2): folio-foto.jpg if (sheetData[i][2] === "No" || sheetData[i][2] === "") { sheet.getRange(rowIndex, 3).setValue("Si"); } // Columna E (Índice 4): Procesado por var procesado = sheetData[i][4]; if (procesado === "") { sheet.getRange(rowIndex, 5).setValue("Pedro"); } else if (procesado.indexOf("Pedro") === -1) { sheet.getRange(rowIndex, 5).setValue(procesado + "/Pedro"); } // Columna G (Índice 6): Observaciones var obs = sheetData[i][6]; if (obs === "") { sheet.getRange(rowIndex, 7).setValue("Imagenes procesadas"); } else { sheet.getRange(rowIndex, 7).setValue(obs + "//Imagenes procesadas"); } // Al retornar aquí, aseguramos que solo modifique el primer folio duplicado que encuentre return jsonOutput({ status: "success", message: "Hoja actualizada" }); } } return jsonOutput({ status: "error", message: "Folio no encontrado en la hoja" }); } // ORIGINAL: Subida de imagen a Drive if (data.base64 && data.fileName) { var folder = DriveApp.getFolderById(FOLDER_ID); var bytes = Utilities.base64Decode(data.base64); var blob = Utilities.newBlob(bytes, data.mimeType || "image/jpeg", data.fileName); var file = folder.createFile(blob); return jsonOutput({ status: "success", fileUrl: file.getUrl(), fileName: file.getName() }); } return jsonOutput({ status: "error", message: "Acción no reconocida o parámetros insuficientes" }); } catch (error) { return jsonOutput({ status: "error", message: error.toString() }); } } // ---------------------------------------------------- // HELPER // ---------------------------------------------------- function jsonOutput(obj) { return ContentService.createTextOutput(JSON.stringify(obj)) .setMimeType(ContentService.MimeType.JSON); }
